create or replace function public.set_updated_at() returns trigger
language plpgsql as $$
begin
  new.updated_at = now();
  return new;
end $$;

create or replace function public.is_service_role() returns boolean
language sql stable as $$
  -- service-role API key, OR a direct trusted DB session (migrations / seed) that carries no end-user JWT.
  -- NOTE: uses session_user (not current_user) so SECURITY DEFINER functions cannot spoof it.
  select coalesce(nullif(current_setting('request.jwt.claims', true), '')::jsonb ->> 'role', '') = 'service_role'
      or (session_user in ('postgres','supabase_admin','supabase_auth_admin')
          and coalesce(nullif(current_setting('request.jwt.claims', true), ''), '') = '');
$$;

-- Generic row audit trigger
create or replace function public.audit_row() returns trigger
language plpgsql security definer set search_path = public as $$
declare v_id text;
begin
  v_id := coalesce(to_jsonb(new) ->> 'id', to_jsonb(old) ->> 'id',
                   to_jsonb(new) ->> 'user_id', to_jsonb(old) ->> 'user_id');
  insert into public.audit_logs(actor_id, action, entity_type, entity_id, metadata)
  values (auth.uid(), lower(tg_op), tg_table_name, v_id,
          case when tg_op = 'UPDATE' then jsonb_build_object('status_before', to_jsonb(old) ->> 'status', 'status_after', to_jsonb(new) ->> 'status')
               when tg_op = 'DELETE' then jsonb_build_object('deleted', true)
               else '{}'::jsonb end);
  return coalesce(new, old);
end $$;

-- ---------------------------------------------------------------------
-- New auth user -> profile + SAFE default role
-- Only 'student' or 'parent' can ever be chosen at signup. Everything else is ignored.
-- ---------------------------------------------------------------------
create or replace function public.handle_new_user() returns trigger
language plpgsql security definer set search_path = public as $$
declare
  v_type text := coalesce(new.raw_user_meta_data ->> 'account_type', 'student');
  v_role public.app_role;
  v_locale text := coalesce(new.raw_user_meta_data ->> 'locale', 'en');
begin
  v_role := case when v_type = 'parent' then 'parent'::public.app_role else 'student'::public.app_role end;
  insert into public.profiles(id, full_name, preferred_locale)
  values (new.id,
          left(coalesce(new.raw_user_meta_data ->> 'full_name', ''), 120),
          case when v_locale in ('en','ar') then v_locale else 'en' end);
  insert into public.user_roles(user_id, role) values (new.id, v_role);
  return new;
end $$;

create trigger on_auth_user_created after insert on auth.users
for each row execute function public.handle_new_user();

-- Protect sensitive profile columns from self-modification
create or replace function public.guard_profile_update() returns trigger
language plpgsql security definer set search_path = public as $$
begin
  if public.is_service_role() or public.is_admin() then return new; end if;
  if new.account_status is distinct from old.account_status
     or new.family_link_code is distinct from old.family_link_code
     or new.is_test_data is distinct from old.is_test_data
     or new.id is distinct from old.id then
    raise exception 'not allowed to change protected profile fields' using errcode = '42501';
  end if;
  return new;
end $$;
create trigger profiles_guard before update on public.profiles for each row execute function public.guard_profile_update();

-- Role changes: only admins; only super admins may grant/revoke admin or super_admin
create or replace function public.guard_user_roles() returns trigger
language plpgsql security definer set search_path = public as $$
declare v_role public.app_role := coalesce(new.role, old.role);
begin
  if public.is_service_role() then return coalesce(new, old); end if;
  if not public.is_admin() then
    raise exception 'only administrators can change roles' using errcode = '42501';
  end if;
  if v_role in ('admin','super_admin') and not public.is_super_admin() then
    raise exception 'only super administrators can manage admin roles' using errcode = '42501';
  end if;
  if tg_op = 'DELETE' and old.user_id = auth.uid() and old.role in ('admin','super_admin') then
    raise exception 'you cannot remove your own administrator role' using errcode = '42501';
  end if;
  if tg_op = 'INSERT' then new.granted_by := auth.uid(); end if;
  return coalesce(new, old);
end $$;
create trigger user_roles_guard before insert or update or delete on public.user_roles
for each row execute function public.guard_user_roles();
create trigger user_roles_audit after insert or update or delete on public.user_roles
for each row execute function public.audit_row();

create or replace function public.coordinates_student(student uuid, uid uuid default auth.uid()) returns boolean
language sql stable security definer set search_path = public as $$
  select exists (
    select 1 from public.organization_members s
    join public.organization_members c on c.organization_id = s.organization_id

language sql stable security definer set search_path = public as $$
  select exists (select 1 from public.parent_student_links
                 where parent_id = uid and student_id = student and status = 'verified')
     and public.has_role('parent', uid);
$$;

-- Parent requests a link with the student's family code. Admin verifies.
create or replace function public.request_parent_link(p_code text) returns uuid
language plpgsql security definer set search_path = public as $$
declare v_student uuid; v_id uuid;
begin
  if not public.has_role('parent') then raise exception 'only parent accounts can request links' using errcode='42501'; end if;
  select p.id into v_student from public.profiles p
    join public.user_roles r on r.user_id = p.id and r.role = 'student'
    where p.family_link_code = upper(trim(p_code));
  if v_student is null then raise exception 'invalid code' using errcode='P0002'; end if;
  insert into public.parent_student_links(parent_id, student_id) values (auth.uid(), v_student)
  on conflict (parent_id, student_id) do update set status = case when parent_student_links.status in ('rejected','revoked') then 'pending' else parent_student_links.status end
  returning id into v_id;
  perform public.write_audit('request','parent_student_links', v_id::text);
  return v_id;
end $$;

create or replace function public.guard_parent_links() returns trigger
language plpgsql security definer set search_path = public as $$
begin
  if public.is_service_role() or public.is_admin() then
    if new.status = 'verified' and old.status is distinct from 'verified' then
      new.verified_by := auth.uid(); new.verified_at := now();
    end if;
    return new;
  end if;
  raise exception 'only administrators can change family links' using errcode='42501';
end $$;
create trigger psl_guard before update on public.parent_student_links for each row execute function public.guard_parent_links();
create trigger psl_audit after insert or update or delete on public.parent_student_links for each row execute function public.audit_row();

-- Enrollment is created ONLY through this function (no direct insert policy for students).
create or replace function public.enroll_in_course(p_course uuid, p_student uuid default null) returns public.enrollments
language plpgsql security definer set search_path = public as $$
declare
  c public.courses; v_student uuid := coalesce(p_student, auth.uid());
  v_count int; v_status public.enrollment_status; v_source text := 'self'; e public.enrollments;
begin
  if auth.uid() is null then raise exception 'authentication required' using errcode='42501'; end if;
  if not public.is_active_user() then raise exception 'account is not active' using errcode='42501'; end if;
  if v_student <> auth.uid() then
    if public.is_parent_of(v_student) then v_source := 'parent';
    elsif public.is_admin() then v_source := 'admin';
    else raise exception 'not allowed to enroll this student' using errcode='42501'; end if;
  end if;
  if not public.has_role('student', v_student) then raise exception 'only student accounts can be enrolled' using errcode='42501'; end if;

  select * into c from public.courses where id = p_course for update;
  if c.id is null or c.status <> 'published' then raise exception 'course not available' using errcode='P0002'; end if;
  if c.availability not in ('open') and v_source <> 'admin' then raise exception 'enrollment is not open for this course' using errcode='P0001'; end if;

  select * into e from public.enrollments where course_id = p_course and student_id = v_student;
  if e.id is not null and e.status not in ('cancelled','refunded') then return e; end if;

  if c.capacity is not null then
    select count(*) into v_count from public.enrollments where course_id = p_course and status in ('active','completed','pending_payment','pending_approval');
    if v_count >= c.capacity then raise exception 'course is full' using errcode='P0001'; end if;
  end if;

  v_status := case when v_source = 'admin' then 'active'
                   when c.price_amount > 0 then 'pending_payment'
                   when c.requires_approval then 'pending_approval'
                   else 'active' end;

  if e.id is not null then
    update public.enrollments set status = v_status, source = v_source, requested_by = auth.uid(),
      activated_at = case when v_status = 'active' then now() end, enrolled_at = now()
      where id = e.id returning * into e;
  else
    insert into public.enrollments(course_id, student_id, status, source, requested_by, activated_at)
    values (p_course, v_student, v_status, v_source, auth.uid(), case when v_status = 'active' then now() end)
    returning * into e;
  end if;
  return e;
end $$;
revoke execute on function public.enroll_in_course(uuid, uuid) from anon;

-- Only staff may change enrollment status directly (payments go through service role)
create or replace function public.guard_enrollment_update() returns trigger
language plpgsql security definer set search_path = public as $$
begin
  if public.is_service_role() or public.is_admin() then
    if new.status = 'active' and old.status is distinct from 'active' then new.activated_at := coalesce(new.activated_at, now()); end if;
    if new.status = 'completed' and old.status is distinct from 'completed' then new.completed_at := coalesce(new.completed_at, now()); end if;
    return new;
  end if;
  -- trainers may mark completion of their course's enrollments
  if public.is_course_instructor(old.course_id) and new.status = 'completed' and old.status = 'active'
     and new.course_id = old.course_id and new.student_id = old.student_id then
    new.completed_at := now(); return new;
  end if;
  -- students may cancel their own unpaid request
  if old.student_id = auth.uid() and new.status = 'cancelled' and old.status in ('pending_payment','pending_approval')
     and new.course_id = old.course_id and new.student_id = old.student_id then
    return new;
  end if;
  raise exception 'not allowed to change this enrollment' using errcode='42501';
end $$;
create trigger enrollments_guard before update on public.enrollments for each row execute function public.guard_enrollment_update();

-- Start (or resume) an attempt
create or replace function public.start_attempt(p_assessment uuid) returns public.assessment_attempts
language plpgsql security definer set search_path = public as $$
declare a public.assessments; v_n int; t public.assessment_attempts;
begin
  select * into a from public.assessments where id = p_assessment and status = 'published';
  if a.id is null then raise exception 'assessment not found' using errcode='P0002'; end if;
  if not public.is_enrolled(a.course_id) then raise exception 'you are not enrolled in this course' using errcode='42501'; end if;
  select * into t from public.assessment_attempts where assessment_id = a.id and student_id = auth.uid() and status = 'in_progress' limit 1;
  if t.id is not null then return t; end if;
  select count(*) into v_n from public.assessment_attempts where assessment_id = a.id and student_id = auth.uid();
  if a.max_attempts is not null and v_n >= a.max_attempts then raise exception 'no attempts remaining' using errcode='P0001'; end if;
  insert into public.assessment_attempts(assessment_id, student_id, course_id, attempt_number)
  values (a.id, auth.uid(), a.course_id, v_n + 1) returning * into t;
  return t;
end $$;

-- Submit + grade. p_answers: [{"question_id": "...", "option_ids": ["..."], "text": "..."}]
create or replace function public.submit_attempt(p_attempt uuid, p_answers jsonb) returns public.assessment_attempts
language plpgsql security definer set search_path = public as $$
declare
  t public.assessment_attempts; a public.assessments; q record; k public.question_keys;
  v_ans jsonb; v_sel uuid[]; v_text text; v_ok boolean; v_pts numeric; v_total numeric := 0; v_max numeric := 0; v_review boolean := false;
begin
  select * into t from public.assessment_attempts where id = p_attempt for update;
  if t.id is null or t.student_id <> auth.uid() then raise exception 'attempt not found' using errcode='P0002'; end if;
  if t.status <> 'in_progress' then raise exception 'attempt already submitted' using errcode='P0001'; end if;
  select * into a from public.assessments where id = t.assessment_id;
  if a.time_limit_minutes is not null and now() > t.started_at + make_interval(mins => a.time_limit_minutes + 2) then
    -- late: still graded, but recorded (2 minute network grace)
    null;
  end if;

  for q in select * from public.questions where assessment_id = a.id order by position loop
    v_max := v_max + q.points;
    select x into v_ans from jsonb_array_elements(coalesce(p_answers,'[]'::jsonb)) x where x ->> 'question_id' = q.id::text limit 1;
    v_sel := coalesce((select array_agg(value::uuid) from jsonb_array_elements_text(coalesce(v_ans -> 'option_ids','[]'::jsonb))), '{}');
    -- only accept options that belong to this question
    v_sel := coalesce((select array_agg(o.id order by o.id) from public.answer_options o where o.question_id = q.id and o.id = any(v_sel)), '{}');
    v_text := left(v_ans ->> 'text', 4000);
    select * into k from public.question_keys where question_id = q.id;
    v_ok := null; v_pts := null;
    if q.kind in ('single_choice','multiple_choice') then
      v_ok := k.question_id is not null
              and coalesce(array_length(v_sel,1),0) > 0
              and (select coalesce(array_agg(x order by x),'{}') from unnest(k.correct_option_ids) x) = v_sel;
      v_pts := case when v_ok then q.points else 0 end;
    else
      if k.question_id is not null and coalesce(array_length(k.accepted_answers,1),0) > 0 then
        v_ok := exists (select 1 from unnest(k.accepted_answers) acc
                        where case when k.case_sensitive then trim(acc) = trim(coalesce(v_text,''))
                                   else lower(trim(acc)) = lower(trim(coalesce(v_text,''))) end);
        v_pts := case when v_ok then q.points else 0 end;
      else
        v_review := true; -- manual grading workflow
      end if;
    end if;
    insert into public.assessment_answers(attempt_id, question_id, selected_option_ids, text_answer, is_correct, points_awarded)
    values (t.id, q.id, v_sel, v_text, v_ok, v_pts)
    on conflict (attempt_id, question_id) do update set selected_option_ids = excluded.selected_option_ids,
      text_answer = excluded.text_answer, is_correct = excluded.is_correct, points_awarded = excluded.points_awarded;
    v_total := v_total + coalesce(v_pts, 0);
  end loop;

  update public.assessment_attempts set
    submitted_at = now(),
    status = case when v_review then 'needs_review' else 'graded' end,
    score_points = v_total, max_points = v_max,
    score_percent = case when v_max > 0 then round(v_total * 100 / v_max, 2) else 0 end,
    passed = case when v_review then null else (case when v_max > 0 then v_total * 100 / v_max else 0 end) >= a.pass_percent end
  where id = t.id returning * into t;
  return t;
end $$;

-- Trainer/admin finishes a manual review
create or replace function public.review_attempt(p_attempt uuid, p_points jsonb, p_feedback text default null) returns public.assessment_attempts
language plpgsql security definer set search_path = public as $$
declare t public.assessment_attempts; a public.assessments; v_total numeric;
begin
  select * into t from public.assessment_attempts where id = p_attempt for update;
  if t.id is null then raise exception 'attempt not found' using errcode='P0002'; end if;
  if not public.is_course_staff(t.course_id) then raise exception 'not allowed' using errcode='42501'; end if;
  select * into a from public.assessments where id = t.assessment_id;
  update public.assessment_answers aa set
    points_awarded = least(greatest((p_points ->> aa.question_id::text)::numeric, 0), q.points),
    is_correct = (p_points ->> aa.question_id::text)::numeric >= q.points
  from public.questions q
  where aa.attempt_id = t.id and q.id = aa.question_id and p_points ? aa.question_id::text;
  select coalesce(sum(points_awarded),0) into v_total from public.assessment_answers where attempt_id = t.id;
  update public.assessment_attempts set status = 'graded', score_points = v_total,
    score_percent = case when max_points > 0 then round(v_total * 100 / max_points, 2) else 0 end,
    passed = (case when max_points > 0 then v_total * 100 / max_points else 0 end) >= a.pass_percent,
    reviewed_by = auth.uid(), feedback = left(p_feedback, 4000)
  where id = t.id returning * into t;
  perform public.write_audit('review','assessment_attempts', t.id::text, jsonb_build_object('score', t.score_percent));
  return t;
end $$;

create or replace function public.guard_submission() returns trigger
language plpgsql security definer set search_path = public as $$
declare v_course uuid; v_max int;
begin
  select course_id, max_points into v_course, v_max from public.assignments where id = new.assignment_id;
  new.course_id := v_course;
  if public.is_service_role() then return new; end if;
  if public.is_course_staff(v_course) and (tg_op = 'UPDATE') then
    if new.grade_points is not null and (new.grade_points < 0 or new.grade_points > v_max) then
      raise exception 'grade out of range' using errcode='22003'; end if;
    if new.grade_points is distinct from old.grade_points or new.feedback is distinct from old.feedback then
      new.graded_by := auth.uid(); new.graded_at := now();
      if new.grade_points is not null then new.status := 'graded'; end if;
    end if;
    -- staff cannot rewrite the student's work
    new.body_text := old.body_text; new.link_url := old.link_url; new.file_url := old.file_url; new.student_id := old.student_id;
    return new;
  end if;
  -- student path
  if new.student_id <> auth.uid() or not public.is_enrolled(v_course) then
    raise exception 'not allowed' using errcode='42501'; end if;
  if tg_op = 'UPDATE' then
    if old.status = 'graded' then raise exception 'graded submissions cannot be changed' using errcode='42501'; end if;
    new.student_id := old.student_id;
  end if;
  new.grade_points := case when tg_op = 'UPDATE' then old.grade_points else null end;
  new.feedback := case when tg_op = 'UPDATE' then old.feedback else null end;
  new.graded_by := case when tg_op = 'UPDATE' then old.graded_by else null end;
  new.graded_at := case when tg_op = 'UPDATE' then old.graded_at else null end;
  new.status := 'submitted'; new.submitted_at := now();
  return new;
end $$;
create trigger submissions_guard before insert or update on public.submissions for each row execute function public.guard_submission();
create trigger submissions_audit after update on public.submissions for each row execute function public.audit_row();

-- ---------------------------------------------------------------------
-- Progress rules (the ONLY way a lesson becomes completed)
-- ---------------------------------------------------------------------
create or replace function public.track_lesson_progress(p_lesson uuid, p_video_percent int default null, p_position int default null)
returns public.lesson_progress
language plpgsql security definer set search_path = public as $$
declare l public.lessons; p public.lesson_progress;
begin
  select * into l from public.lessons where id = p_lesson and status = 'published';
  if l.id is null then raise exception 'lesson not found' using errcode='P0002'; end if;
  if not public.is_enrolled(l.course_id) then raise exception 'you are not enrolled in this course' using errcode='42501'; end if;
  insert into public.lesson_progress(student_id, lesson_id, course_id, video_percent, last_position_seconds)
  values (auth.uid(), l.id, l.course_id, least(greatest(coalesce(p_video_percent,0),0),100), greatest(coalesce(p_position,0),0))
  on conflict (student_id, lesson_id) do update set
    -- video percent can only go up (the furthest verified point)
    video_percent = greatest(lesson_progress.video_percent, least(greatest(coalesce(p_video_percent, 0),0),100)),
    last_position_seconds = coalesce(greatest(p_position,0), lesson_progress.last_position_seconds),
    updated_at = now()
  returning * into p;
  return p;
end $$;

create or replace function public.complete_lesson(p_lesson uuid) returns public.lesson_progress
language plpgsql security definer set search_path = public as $$
declare l public.lessons; p public.lesson_progress; v_ok boolean;
begin
  p := public.track_lesson_progress(p_lesson, null, null);
  select * into l from public.lessons where id = p_lesson;
  v_ok := case l.completion_rule
    when 'view' then true
    when 'video_90' then p.video_percent >= 90
    when 'quiz_pass' then exists (select 1 from public.assessment_attempts where assessment_id = l.assessment_id and student_id = auth.uid() and passed)
    when 'assignment_submitted' then exists (select 1 from public.submissions where assignment_id = l.assignment_id and student_id = auth.uid())
    else false end;
  if not v_ok then raise exception 'completion requirements not met (%)', l.completion_rule using errcode='P0001'; end if;
  update public.lesson_progress set status = 'completed', completed_at = coalesce(completed_at, now()), updated_at = now()
  where student_id = auth.uid() and lesson_id = p_lesson returning * into p;
  return p;
end $$;

-- staff manual completion
create or replace function public.staff_complete_lesson(p_lesson uuid, p_student uuid) returns void
language plpgsql security definer set search_path = public as $$
declare l public.lessons;
begin
  select * into l from public.lessons where id = p_lesson;
  if not public.is_course_staff(l.course_id) then raise exception 'not allowed' using errcode='42501'; end if;
  if not public.is_enrolled(l.course_id, p_student) then raise exception 'student not enrolled' using errcode='P0001'; end if;
  insert into public.lesson_progress(student_id, lesson_id, course_id, status, completed_at)
  values (p_student, l.id, l.course_id, 'completed', now())
  on conflict (student_id, lesson_id) do update set status = 'completed', completed_at = coalesce(lesson_progress.completed_at, now());
  perform public.write_audit('staff_complete','lesson_progress', p_lesson::text, jsonb_build_object('student', p_student));
end $$;

create or replace function public.course_progress(p_course uuid, p_student uuid default auth.uid())
returns table (total_lessons int, completed_lessons int, percent int)
language sql stable security definer set search_path = public as $$
  with t as (select count(*)::int n from public.lessons where course_id = p_course and status = 'published'),
       c as (select count(*)::int n from public.lesson_progress lp join public.lessons l on l.id = lp.lesson_id and l.status = 'published'
             where lp.course_id = p_course and lp.student_id = p_student and lp.status = 'completed')
  select t.n, c.n, case when t.n = 0 then 0 else (c.n * 100 / t.n) end from t, c
  where public.can_view_student(p_student) or public.is_course_staff(p_course);
$$;

create or replace function public.generate_certificate_number() returns text
language plpgsql as $$
declare v text; alphabet text := 'ABCDEFGHJKLMNPQRSTUVWXYZ23456789'; i int;
begin
  loop
    v := 'WT-' || to_char(now(),'YYYY') || '-';
    for i in 1..8 loop v := v || substr(alphabet, 1 + (get_byte(extensions.gen_random_bytes(1),0) % 32), 1); end loop;
    exit when not exists (select 1 from public.certificates where certificate_number = v);
  end loop;
  return v;
end $$;

-- Issuance: only course staff, only for an enrollment whose lessons are all completed (or marked completed)
create or replace function public.issue_certificate(p_enrollment uuid) returns public.certificates
language plpgsql security definer set search_path = public as $$
declare e public.enrollments; c public.courses; pr record; cert public.certificates; v_name text;
begin
  select * into e from public.enrollments where id = p_enrollment for update;
  if e.id is null then raise exception 'enrollment not found' using errcode='P0002'; end if;
  if not public.is_course_staff(e.course_id) then raise exception 'not allowed to issue certificates' using errcode='42501'; end if;
  select * into cert from public.certificates where enrollment_id = e.id;
  if cert.id is not null then return cert; end if;
  select * into pr from public.course_progress(e.course_id, e.student_id);
  if e.status <> 'completed' and (pr.total_lessons = 0 or pr.completed_lessons < pr.total_lessons) then
    raise exception 'student has not completed the course' using errcode='P0001'; end if;
  select * into c from public.courses where id = e.course_id;
  select nullif(trim(full_name),'') into v_name from public.profiles where id = e.student_id;
  if v_name is null then raise exception 'student profile has no name' using errcode='P0001'; end if;
  if e.status <> 'completed' then update public.enrollments set status = 'completed' where id = e.id; end if;
  insert into public.certificates(certificate_number, enrollment_id, student_id, course_id, student_name, course_title_en, course_title_ar,
    hours_total, completion_date, issued_by, is_test_data)
  values (public.generate_certificate_number(), e.id, e.student_id, e.course_id, v_name, c.title_en, c.title_ar,
    c.hours_total, current_date, auth.uid(), c.is_test_data)
  returning * into cert;
  perform public.write_audit('issue','certificates', cert.id::text, jsonb_build_object('number', cert.certificate_number));
  insert into public.notifications(user_id, kind, title_en, title_ar, body_en, body_ar, link)
  values (e.student_id, 'certificate_issued', 'Your certificate is ready', 'شهادتك جاهزة',
          'Congratulations on completing ' || c.title_en, 'مبارك إتمام ' || c.title_ar, '/dashboard/certificates');
  return cert;
end $$;

create or replace function public.revoke_certificate(p_id uuid, p_reason text) returns void
language plpgsql security definer set search_path = public as $$
begin
  if not public.is_admin() then raise exception 'only administrators can revoke certificates' using errcode='42501'; end if;
  update public.certificates set status = 'revoked', revoked_reason = left(p_reason, 500) where id = p_id;
  perform public.write_audit('revoke','certificates', p_id::text, jsonb_build_object('reason', p_reason));
end $$;

-- Public verification with minimal personal data
create or replace function public.verify_certificate(p_number text) returns jsonb
language plpgsql security definer set search_path = public as $$
declare c public.certificates; v_num text := upper(trim(coalesce(p_number,'')));
begin
  if v_num !~ '^WT-[0-9]{4}-[A-Z0-9]{8}$' then return jsonb_build_object('found', false); end if;
  select * into c from public.certificates where certificate_number = v_num;
  insert into public.certificate_verifications(certificate_id, queried_number, found) values (c.id, v_num, c.id is not null);
  if c.id is null then return jsonb_build_object('found', false); end if;
  return jsonb_build_object('found', true, 'certificate_number', c.certificate_number, 'student_name', c.student_name,
    'course_title_en', c.course_title_en, 'course_title_ar', c.course_title_ar, 'completion_date', c.completion_date,
    'status', c.status, 'is_test_data', c.is_test_data);
end $$;
grant execute on function public.verify_certificate(text) to anon, authenticated;

create or replace function public.guard_project() returns trigger
language plpgsql security definer set search_path = public as $$
begin
  if public.is_service_role() or public.is_admin() then
    if tg_op = 'UPDATE' and new.approval_status is distinct from old.approval_status then new.reviewed_by := auth.uid(); end if;
    return new;
  end if;
  if new.student_id <> auth.uid() then raise exception 'not allowed' using errcode='42501'; end if;
  -- students can request public showcase but never approve themselves
  if new.visibility = 'public' then
    if tg_op = 'INSERT' or old.approval_status <> 'approved' or new.title is distinct from old.title
       or new.description is distinct from old.description or new.cover_url is distinct from old.cover_url then
      new.approval_status := 'pending';
    end if;
  else
    new.approval_status := 'not_requested';
  end if;
  if tg_op = 'UPDATE' then new.reviewed_by := old.reviewed_by; new.is_test_data := old.is_test_data; else new.is_test_data := false; end if;
  return new;
end $$;

create or replace function public.register_for_event(p_event uuid, p_attendee_name text default null) returns public.event_registrations
language plpgsql security definer set search_path = public as $$
declare ev public.events; n int; r public.event_registrations; v_name text;
begin
  if auth.uid() is null then raise exception 'authentication required' using errcode='42501'; end if;
  select * into ev from public.events where id = p_event for update;
  if ev.id is null or ev.status <> 'published' then raise exception 'event not available' using errcode='P0002'; end if;
  if not ev.registration_open or ev.starts_at < now() then raise exception 'registration is closed' using errcode='P0001'; end if;
  select * into r from public.event_registrations where event_id = ev.id and user_id = auth.uid();
  if r.id is not null and r.status <> 'cancelled' then return r; end if;
  select count(*) into n from public.event_registrations where event_id = ev.id and status = 'registered';
  v_name := coalesce(nullif(trim(p_attendee_name),''), (select full_name from public.profiles where id = auth.uid()), 'Attendee');
  insert into public.event_registrations(event_id, user_id, attendee_name, status)
  values (ev.id, auth.uid(), left(v_name,120), case when ev.capacity is not null and n >= ev.capacity then 'waitlisted' else 'registered' end)
  on conflict (event_id, user_id) do update set status = excluded.status, attendee_name = excluded.attendee_name
  returning * into r;
  insert into public.notifications(user_id, kind, title_en, title_ar, body_en, body_ar, link)
  values (auth.uid(), 'event_registration',
    case when r.status = 'registered' then 'Event registration confirmed' else 'You are on the waitlist' end,
    case when r.status = 'registered' then 'تم تأكيد تسجيلك في الفعالية' else 'أنت على قائمة الانتظار' end,
    ev.title_en, ev.title_ar, '/events/' || ev.slug);
  return r;
end $$;

create or replace function public.guard_team() returns trigger
language plpgsql security definer set search_path = public as $$
declare c public.competitions; n int;
begin
  if public.is_service_role() or public.is_admin() then return new; end if;
  select * into c from public.competitions where id = new.competition_id;
  if new.captain_id <> auth.uid() then raise exception 'not allowed' using errcode='42501'; end if;
  if tg_op = 'INSERT' then
    if c.status <> 'open' or (c.registration_deadline is not null and c.registration_deadline < now()) then
      raise exception 'registration is closed' using errcode='P0001'; end if;
    new.status := 'submitted';
  else
    if new.status not in ('withdrawn', old.status) then raise exception 'status can only be changed by administrators' using errcode='42501'; end if;
  end if;
  n := jsonb_array_length(new.members);
  if n < c.team_min or n > c.team_max then raise exception 'team size must be between % and %', c.team_min, c.team_max using errcode='P0001'; end if;
  return new;
end $$;

-- returns true if allowed; service-role only (called from the server)
create or replace function public.rate_limit_hit(p_bucket text, p_limit int, p_window_seconds int) returns boolean
language plpgsql security definer set search_path = public as $$
declare v_window timestamptz := to_timestamp(floor(extract(epoch from now()) / p_window_seconds) * p_window_seconds); v_hits int;
begin
  insert into public.rate_limits(bucket, window_start, hits) values (p_bucket, v_window, 1)
  on conflict (bucket, window_start) do update set hits = rate_limits.hits + 1
  returning hits into v_hits;
  delete from public.rate_limits where window_start < now() - interval '1 day';
  return v_hits <= p_limit;
end $$;
revoke execute on function public.rate_limit_hit(text,int,int) from public, anon, authenticated;
grant execute on function public.rate_limit_hit(text,int,int) to service_role;

-- Create (or reuse) the order for a pending_payment enrollment. Price is ALWAYS read from the course row.
create or replace function public.create_order_for_enrollment(p_enrollment uuid, p_discount_code text default null) returns public.orders
language plpgsql security definer set search_path = public as $$
declare e public.enrollments; c public.courses; d public.discount_codes; o public.orders; v_disc numeric := 0;
begin
  select * into e from public.enrollments where id = p_enrollment for update;
  if e.id is null then raise exception 'enrollment not found' using errcode='P0002'; end if;
  if not (e.student_id = auth.uid() or public.is_parent_of(e.student_id) or public.is_admin()) then
    raise exception 'not allowed' using errcode='42501'; end if;
  if e.status <> 'pending_payment' then raise exception 'enrollment does not require payment' using errcode='P0001'; end if;
  select * into o from public.orders where enrollment_id = e.id and status = 'pending' order by created_at desc limit 1;
  if o.id is not null and p_discount_code is null then return o; end if;
  select * into c from public.courses where id = e.course_id;
  if p_discount_code is not null and length(trim(p_discount_code)) > 0 then
    select * into d from public.discount_codes where code = upper(trim(p_discount_code)) and active
      and (valid_until is null or valid_until > now()) and (max_uses is null or used_count < max_uses);
    if d.code is null then raise exception 'invalid discount code' using errcode='P0001'; end if;
    v_disc := round(c.price_amount * d.percent_off / 100.0, 3);
  end if;
  if o.id is not null then update public.orders set status = 'cancelled' where id = o.id; end if;
  insert into public.orders(user_id, student_id, course_id, enrollment_id, subtotal, discount_amount, total, currency, discount_code)
  values (auth.uid(), e.student_id, c.id, e.id, c.price_amount, v_disc, greatest(c.price_amount - v_disc, 0), c.currency, d.code)
  returning * into o;
  return o;
end $$;

-- SERVER ONLY: record a verified provider result. Amount & currency must match the order exactly.
create or replace function public.record_payment_result(p_order uuid, p_provider text, p_reference text, p_amount numeric, p_currency public.currency_code,
  p_succeeded boolean, p_result_code text, p_result_description text, p_is_sandbox boolean) returns public.orders
language plpgsql security definer set search_path = public as $$
declare o public.orders;
begin
  if not public.is_service_role() then raise exception 'server only' using errcode='42501'; end if;
  select * into o from public.orders where id = p_order for update;
  if o.id is null then raise exception 'order not found' using errcode='P0002'; end if;
  insert into public.payment_transactions(order_id, provider, type, provider_reference, amount, currency, status, result_code, result_description, is_sandbox)
  values (o.id, p_provider, 'payment', p_reference, p_amount, p_currency, case when p_succeeded then 'succeeded' else 'failed' end,
          left(p_result_code,64), left(p_result_description,300), p_is_sandbox)
  on conflict (provider, provider_reference, type) do nothing;
  if o.status = 'paid' then return o; end if; -- idempotent
  if p_succeeded then
    if p_amount <> o.total or p_currency <> o.currency then
      update public.orders set status = 'failed' where id = o.id returning * into o;
      perform public.write_audit('amount_mismatch','orders', o.id::text, jsonb_build_object('expected', o.total, 'got', p_amount));
      return o;
    end if;
    update public.orders set status = 'paid', paid_at = now(), provider = p_provider, is_sandbox = p_is_sandbox where id = o.id returning * into o;
    update public.enrollments set status = case when (select requires_approval from public.courses where id = o.course_id) then 'pending_approval'::public.enrollment_status else 'active'::public.enrollment_status end
      where id = o.enrollment_id and status = 'pending_payment';
    if o.discount_code is not null then update public.discount_codes set used_count = used_count + 1 where code = o.discount_code; end if;
  else
    update public.orders set status = 'failed' where id = o.id and status = 'pending' returning * into o;
  end if;
  return o;
end $$;
revoke execute on function public.record_payment_result(uuid,text,text,numeric,public.currency_code,boolean,text,text,boolean) from public, anon, authenticated;
grant execute on function public.record_payment_result(uuid,text,text,numeric,public.currency_code,boolean,text,text,boolean) to service_role;

-- Admin refund status (provider refund call happens in the server adapter first)
create or replace function public.mark_order_refunded(p_order uuid, p_amount numeric, p_reference text) returns public.orders
language plpgsql security definer set search_path = public as $$
declare o public.orders;
begin
  if not (public.is_service_role() or public.is_admin()) then raise exception 'not allowed' using errcode='42501'; end if;
  select * into o from public.orders where id = p_order for update;
  if o.status not in ('paid','partially_refunded') then raise exception 'order is not paid' using errcode='P0001'; end if;
  if p_amount <= 0 or p_amount > o.total then raise exception 'invalid refund amount' using errcode='P0001'; end if;
  insert into public.payment_transactions(order_id, provider, type, provider_reference, amount, currency, status, is_sandbox)
  values (o.id, o.provider, 'refund', coalesce(p_reference, 'manual-' || extensions.gen_random_uuid()), p_amount, o.currency, 'succeeded', o.is_sandbox);
  update public.orders set status = case when p_amount = o.total then 'refunded' else 'partially_refunded' end where id = o.id returning * into o;
  if o.status = 'refunded' then update public.enrollments set status = 'refunded' where id = o.enrollment_id; end if;
  return o;
end $$;

-- ADMIN: list users with email + roles
create or replace function public.admin_list_users(p_search text default null, p_role public.app_role default null, p_limit int default 50, p_offset int default 0)
returns table (id uuid, email text, full_name text, account_status public.account_status, roles public.app_role[], created_at timestamptz, last_sign_in_at timestamptz, email_confirmed boolean, is_test_data boolean, total bigint)
language plpgsql stable security definer set search_path = public as $$
begin
  if not public.is_admin() then raise exception 'not allowed' using errcode='42501'; end if;
  return query
  with base as (
    select u.id, u.email::text, p.full_name, p.account_status,
           coalesce(array_agg(r.role order by r.role) filter (where r.role is not null), '{}') as roles,
           u.created_at, u.last_sign_in_at, u.email_confirmed_at is not null as email_confirmed, p.is_test_data
    from auth.users u join public.profiles p on p.id = u.id
    left join public.user_roles r on r.user_id = u.id
    where (p_search is null or u.email ilike '%' || p_search || '%' or p.full_name ilike '%' || p_search || '%')
    group by u.id, p.id
  )
  select b.*, count(*) over () from base b
  where p_role is null or p_role = any(b.roles)
  order by b.created_at desc limit least(p_limit, 200) offset greatest(p_offset, 0);
end $$;

-- School coordinator dashboard (tenant-scoped)
create or replace function public.school_overview(p_org uuid) returns jsonb
language plpgsql stable security definer set search_path = public as $$
begin
  if not (public.is_org_coordinator(p_org) or public.is_admin()) then raise exception 'not allowed' using errcode='42501'; end if;
  return jsonb_build_object(
    'students', (select coalesce(jsonb_agg(s order by s.full_name), '[]'::jsonb) from (
        select p.id, p.full_name,
          (select count(*) from public.enrollments e where e.student_id = p.id and e.status in ('active','completed')) as enrollments,
          (select round(avg(cp.percent)) from public.enrollments e cross join lateral public.course_progress(e.course_id, p.id) cp
             where e.student_id = p.id and e.status in ('active','completed')) as avg_progress,
          (select round(100.0 * count(*) filter (where a.status in ('present','late')) / nullif(count(*),0)) from public.attendance a where a.student_id = p.id) as attendance_rate,
          (select count(*) from public.certificates c where c.student_id = p.id and c.status = 'valid') as certificates
        from public.organization_members m join public.profiles p on p.id = m.user_id
        where m.organization_id = p_org and m.member_role = 'student') s),
    'trainers', (select coalesce(jsonb_agg(jsonb_build_object('id', p.id, 'full_name', p.full_name)), '[]'::jsonb)

        from (select e.course_id, count(*) n from public.enrollments e join public.organization_members m on m.user_id = e.student_id
              and m.organization_id = p_org and m.member_role = 'student' where e.status in ('active','completed') group by e.course_id) x
        join public.courses c on c.id = x.course_id)
  );
end $$;
